cmake_minimum_required(VERSION 3.24)
project(ACSH5PViewer VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(viewer_core STATIC native/viewer.cpp)
target_compile_definitions(viewer_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(viewer_core PUBLIC native "vendor/opus-sdk/headers" vendor/json vendor/webview2/build/native/include)
target_compile_options(viewer_core PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(viewer_core PUBLIC "${CMAKE_SOURCE_DIR}/vendor/webview2/build/native/x64/WebView2LoaderStatic.lib" ole32 oleaut32 shlwapi shell32 user32 comctl32 advapi32 version)
add_library(ACSH5PViewer SHARED native/plugin.cpp native/version.rc)
target_link_libraries(ACSH5PViewer PRIVATE viewer_core)
add_executable(ACSH5PTestHost WIN32 native/host.cpp native/version.rc)
target_link_libraries(ACSH5PTestHost PRIVATE viewer_core)
foreach(target ACSH5PViewer ACSH5PTestHost)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
endforeach()
